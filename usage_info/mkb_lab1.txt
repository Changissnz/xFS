Mob Killer Bot, although serving purposes of interest in the area of computation, is quite 
self-explanatory. Here is the base code for three trials. 

########################################### CODE 

from graph_problems.mkb_ext import * 
from graph_problems.sim_solution_search import * 
from morebs2.numerical_generator import prg__LCG 
import time 

num_agents = 1000 
prg = prg__LCG(44,3,1.11,666.77) 
antimob_score = 500000 
mob_agent_uniform_score = 50000 
mg = MKBot.generate_instance(num_agents,prg,antimob_score,mob_agent_uniform_score)

prngs = prg_to_prg__LCG_sequence(prg,30,4.141414)
prngs2 = prg_to_prg__LCG_sequence(prngs[-1],30,7.75757575) 
prngs.extend(prngs2)
prngs3 = prg_to_prg__LCG_sequence(prngs[-2],30,7.75757575) 
prngs.extend(prngs3) 

for i in range(90): 
    j0 = int(prg()) % len(prngs) 
    j1 = int(prg()) % len(prngs) 
    p0,p1 = prngs[j0],prngs[j1] 
    prngs_ = merge_two_prgs_into_LCG_sequence(p0,p1,2,[1.010101,4.5454545]) 
    prngs.extend(prngs_)

neg_to_zero = True 
S = SimulationSolutionSearch(mg,MKB_env_run,prngs,\
    MKB_env_prng_assignment_function,MKB_env_mode_shift_function,\
    MKB_env_solution_fetch_function_(neg_to_zero),\
    MKB_env_cmp_solution_(neg_to_zero),verbose=True)

t = time.time() 

while not S.fin_stat: 
    S.process_one()

print("runtime: ",time.time() - t) 

q = np.array([r[2][0] for r in S.results])
q2 = np.array([sum(list(r[2][1].values())) for r in S.results])

mx0 = np.mean(q)
mx1 = np.mean(q2) 
mx2 = np.mean(q - q2)

best = S.best[2] 
best = best[0] - sum(best[1].values())

print("mean anti-mob score: {}".format(mx0)) 
print("mean mob score: {}".format(mx1)) 
print("mean game score: {}".format(mx2)) 
print("best score: {}".format(best))
