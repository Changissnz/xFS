This lab focuses on a generated graph of 13 agents. Agent 6 is the autonomous principal. 
Below is code that runs this Slander Net Bot. 

#-----------------------------------------------------------------------------------------
from graph_problems.snb_ext import * 

# declare the <MutableCEAgentNetwork> base 
num_agents = 13 
prg_state_shape = 6 
r_conn_range = [0.3,0.4] 
s_conn_range = [0.2,0.4] 
t_conn_range = [0.2,0.4] 
s_port_variance_range = [0.,1.]
prg = prg__LCG(45.66,8.94,-999.76,1032.66)

mcan = MutableCEAgentNetwork.generate_instance__type_prng(num_agents,prg_state_shape,r_conn_range,\
    s_conn_range,t_conn_range,s_port_variance_range,prg)

# declare SNBot 
snbot = SNBot.from_MutableCEAgentNetwork(mcan,set([]),False)

# set different PRNGs for every agent 
prg_ = deepcopy(prg)
prg2 = [default_std_Python_prng(abs(round(prg_()))) for i in range(13)] 
for i in range(num_agents): 
    if i == 6: continue 
    snbot.set_agent_prg(i,prg2[i])

# run the SNBot for 50 iterations 
for _ in range(50): 
    next(mcan) 

q = sorted([(k,v.score) for k,v in mcan.cea_map.items()],key=lambda x:x[1]) 
for q_ in q: print(q)
