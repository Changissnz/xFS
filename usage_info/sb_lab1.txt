This lab consists of 9 trials showing the performance of a specific <StrangleBot> instance. 

Below is the base code. 

############################### CODE 

from quant.strangle_graph import * 
from morebs2.numerical_generator import * 

prg = prg__LCG(34.4,-14.31,55.7,505.11)

info_mode = ? 
strangler_energy = ? 
strangle_subject_energy = ? 
enable_consumption = ? 
max_timestamps = ? 

se = StrangleEnv.generate_instance("random",info_mode,prg,strangler_energy,strangle_subject_energy,\
    enable_consumption=enable_consumption) 

se.verbose = True 

prgs = prg_to_prg__LCG_sequence(prg,2,5.10101) 
se.set_prng_for_subject(prgs[0],prgs[1]) 

while not se.fin_stat and se.timestamp < max_timestamps:  
    next(se) 

print("Strangler score")
print(se.strangler.score(se.node_weights))
print("Strangler highest score")
print(se.strangler.highest_score) 

print("Strangle Subject energy")
print(se.strangle_subject.energy)

print("Finished timestamp: ",se.timestamp)
